# 单词本添加单词 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 开发时在单词本追加一条词到清单指向的课次 JSON，并立刻进入内存词库。

**Architecture:** `src/utils/wordAppend.ts` 负责校验、按清单定位文件、把一条词追加进数组。`vite/wordSavePlugin.ts` 只在 Vite `serve` 时接受 `PUT /__words`，按请求顺序读写文件。页面成功后调用 store 的 `appendWord`，不重新加载词库，也不改当前练习和错题计数。

**Tech Stack:** Vue 3、TypeScript、Vite、Pinia、Element Plus、Vitest

**Spec:** `docs/superpowers/specs/2026-11-18-add-word-design.md`

## Global Constraints

- 只在 `npm run dev` 可写。打包后没有按钮，也没有接口
- 当前册、单元、课都是具体课时，写入这一课。任一层是「全部」时，在对话框里再选，选项不含「全部」
- 课的范围与单词本课次下拉相同，包含带词表的单元页（如「单元末-词语之泉」）
- 中文、假名、手写必填。说明、词性可空，写入时用 `""`
- 同一课已有相同中文和假名时，仍然追加
- `PUT /__words`。服务器按清单定位文件，只在数组末尾追加一条，不写 `bookId`、`unit`、`lesson`
- 成功后只把这一条推进 `allWords`。不改当前筛选，不重抽已抽出的练习题，不改 `wordErrorCounts`
- 组件继续用 `<script setup lang="ts">`。有业务含义的函数写简短中文注释，格式对齐 `src/utils/wordDisplay.ts`

```ts
import type { WordsManifest } from '../types/word.ts'
import { wordPages } from './unitPages.ts'

/** 开发服务器追加单词的地址。打包后的页面没有这个接口。 */
export const WORD_SAVE_PATH = '/__words'

export const WORD_APPEND_EMPTY = '必填为空'
export const WORD_APPEND_MISSING = '清单里没有这一课'
export const WORD_APPEND_PATH = '路径不合法'
export const WORD_APPEND_PARSE = '词表无法解析'
export const WORD_APPEND_SHAPE = '词表不是数组'
export const WORD_APPEND_WRITE = '写入失败'
export const WORD_APPEND_FAILED = '添加失败'

const MAX_LESSON_NUMBER = 999

export interface StoredWordFields {
  cn: string
  jp: string
  jp_write: string
  describe: string
  part_of_speech: string
}

export interface WordAppendBody extends StoredWordFields {
  bookId: string
  unit: number
  lesson: number
}

export type WordAppendResult<T> = { ok: true; value: T } | { ok: false; message: string }

/**
 * 校验添加单词的请求。
 * 五个文本字段必须是字符串；中文、假名、手写去空白后不能为空。说明和词性空白写成空字符串。
 *
 * @param raw 请求体
 * @return 去过空白的请求；字段不合法时带失败原因
 */
export function parseWordAppendBody(raw: unknown): WordAppendResult<WordAppendBody> {
  if (!raw || typeof raw !== 'object') return fail(WORD_APPEND_EMPTY)
  const body = raw as Record<string, unknown>
  const cn = trimmed(body.cn)
  const jp = trimmed(body.jp)
  const jpWrite = trimmed(body.jp_write)
  const describe = trimmed(body.describe)
  const partOfSpeech = trimmed(body.part_of_speech)
  if (!cn || !jp || !jpWrite || describe === null || partOfSpeech === null) return fail(WORD_APPEND_EMPTY)
  if (!isLessonNumber(body.unit) || !isLessonNumber(body.lesson) || typeof body.bookId !== 'string' || !body.bookId) {
    return fail(WORD_APPEND_MISSING)
  }
  return {
    ok: true,
    value: {
      bookId: body.bookId,
      unit: body.unit,
      lesson: body.lesson,
      cn,
      jp,
      jp_write: jpWrite,
      describe,
      part_of_speech: partOfSpeech,
    },
  }
}

/**
 * 判断清单里的 file 能不能作为词库相对路径。
 * 拒绝绝对路径、空段和 `..`，避免写到词库目录外面。
 *
 * @param file 清单中的 file
 * @return 可以拼到 public/data/words 下面时为 true
 */
export function isSafeManifestFile(file: string): boolean {
  if (!file || file.startsWith('/') || file.startsWith('\\') || /^[A-Za-z]:/.test(file)) return false
  const parts = file.split(/[/\\]/)
  return parts.every((part) => part.length > 0 && part !== '.' && part !== '..')
}

/**
 * 判断 path.relative 的结果是否仍在词库目录内。
 * 开发服务器算出相对路径后调用，拦住解析后跑出 public/data/words 的位置。
 *
 * @param fromRoot path.relative 得到的路径
 * @return 仍在词库目录内时为 true
 */
export function isInsideWordsRoot(fromRoot: string): boolean {
  if (!fromRoot || fromRoot.startsWith('/') || fromRoot.startsWith('\\') || /^[A-Za-z]:/.test(fromRoot)) return false
  return !fromRoot.split(/[/\\]/).includes('..')
}

/**
 * 按册、单元、课号找到清单里的词表文件。
 * 先匹配课，没有再匹配带词表的单元页。课和页课号相同时用课的文件。
 *
 * @param manifest 词库清单
 * @param bookId 册代号
 * @param unit 单元号
 * @param lesson 课号
 * @return 相对 public/data/words 的 file；找不到或路径不合法时带失败原因
 */
export function resolveWordFile(
  manifest: WordsManifest,
  bookId: string,
  unit: number,
  lesson: number,
): WordAppendResult<string> {
  const book = manifest.books.find((item) => item.code === bookId)
  const unitMeta = book?.units.find((item) => item.unit === unit)
  if (!unitMeta) return fail(WORD_APPEND_MISSING)
  const lessonFile = unitMeta.lessons.find((item) => item.lesson === lesson)?.file
  const pageFile = wordPages(unitMeta).find((item) => item.lesson === lesson)?.file
  const file = lessonFile ?? pageFile
  if (!file) return fail(WORD_APPEND_MISSING)
  if (!isSafeManifestFile(file)) return fail(WORD_APPEND_PATH)
  return { ok: true, value: file }
}

/**
 * 把一条词加到词表数组末尾。
 * 已有条目原样保留。新对象只有五个字段。
 *
 * @param existing 课次 JSON 解析结果
 * @param word 去过空白的新词
 * @return 追加后的数组；现有内容不是数组时失败
 */
export function appendStoredWord(existing: unknown, word: StoredWordFields): WordAppendResult<unknown[]> {
  if (!Array.isArray(existing)) return fail(WORD_APPEND_SHAPE)
  return {
    ok: true,
    value: [
      ...existing,
      {
        cn: word.cn,
        jp: word.jp,
        jp_write: word.jp_write,
        describe: word.describe,
        part_of_speech: word.part_of_speech,
      },
    ],
  }
}

/**
 * 把词表格式化成和现有课次文件一样的 JSON。
 *
 * @param words 追加后的数组
 * @return 4 空格缩进、末尾换行的文本
 */
export function formatWordFile(words: unknown[]): string {
  return `${JSON.stringify(words, null, 4)}\n`
}

/**
 * 从清单和课次文本算出要写回的文件。
 * 文件不存在时从空数组开始。文本不是 JSON 或不为数组时不产生写入内容。
 *
 * @param manifest 词库清单
 * @param existingText 现有文件文本；文件不存在时为 null
 * @param raw 请求体
 * @return 相对路径和完整文件内容
 */
export function prepareWordAppend(
  manifest: WordsManifest,
  existingText: string | null,
  raw: unknown,
): WordAppendResult<{ relativeFile: string; contents: string }> {
  const parsed = parseWordAppendBody(raw)
  if (!parsed.ok) return parsed
  const located = resolveWordFile(manifest, parsed.value.bookId, parsed.value.unit, parsed.value.lesson)
  if (!located.ok) return located
  let existing: unknown = []
  if (existingText !== null) {
    try {
      existing = JSON.parse(existingText) as unknown
    } catch {
      return fail(WORD_APPEND_PARSE)
    }
  }
  const appended = appendStoredWord(existing, parsed.value)
  if (!appended.ok) return appended
  return { ok: true, value: { relativeFile: located.value, contents: formatWordFile(appended.value) } }
}

function fail(message: string): WordAppendResult<never> {
  return { ok: false, message }
}

function trimmed(value: unknown): string | null {
  if (typeof value !== 'string') return null
  return value.trim()
}

function isLessonNumber(value: unknown): value is number {
  return typeof value === 'number' && Number.isInteger(value) && value >= 1 && value <= MAX_LESSON_NUMBER
}
```

```ts
import { mkdir, readFile, writeFile } from 'node:fs/promises'
import type { IncomingMessage, ServerResponse } from 'node:http'
import { dirname, relative, resolve } from 'node:path'
import type { Plugin } from 'vite'
import type { WordsManifest } from '../src/types/word.ts'
import {
  WORD_APPEND_PATH,
  WORD_APPEND_WRITE,
  WORD_SAVE_PATH,
  isInsideWordsRoot,
  prepareWordAppend,
} from '../src/utils/wordAppend.ts'

const MAX_BYTES = 32 * 1024

let writeQueue = Promise.resolve()

/**
 * 让词表写入按提交顺序进行。
 * 两次添加同时读到旧文件时，后写完的一次会盖掉先追加的那条。
 *
 * @param work 一次完整的读、追加和写回
 * @return work 的结果
 */
export function enqueueWordWrite<T>(work: () => Promise<T>): Promise<T> {
  const run = writeQueue.then(work, work)
  writeQueue = run.then(
    () => undefined,
    () => undefined,
  )
  return run
}

/**
 * 开发时把一条词追加进清单指向的课次 JSON。
 * 只在 vite dev 注册，打包后的静态站点没有这个接口。
 */
export function wordSavePlugin(): Plugin {
  return {
    name: 'word-save',
    apply: 'serve',
    configureServer(server) {
      server.middlewares.use((req, res, next) => {
        const pathname = req.url?.split('?')[0]
        if (pathname !== WORD_SAVE_PATH || req.method !== 'PUT') {
          next()
          return
        }
        void enqueueWordWrite(async () => {
          const rawBody = await readBody(req)
          const result = await persistWordAppend(server.config.root, rawBody)
          reply(res, result.status, result.body)
        }).catch(() => reply(res, 400, JSON.stringify({ message: WORD_APPEND_WRITE })))
      })
    },
  }
}

/**
 * 按清单把请求追加到词表文件。
 * 路径必须留在 public/data/words 内。文件缺失时新建；不是数组或无法解析时不改文件。
 *
 * @param root 项目根目录
 * @param rawBody 请求 JSON 文本
 * @return HTTP 状态和响应正文
 */
export async function persistWordAppend(root: string, rawBody: string): Promise<{ status: number; body: string }> {
  let raw: unknown
  try {
    raw = JSON.parse(rawBody) as unknown
  } catch {
    return fail(WORD_APPEND_WRITE)
  }
  let manifest: WordsManifest
  try {
    manifest = JSON.parse(await readFile(resolve(root, 'public', 'data', 'words', 'manifest.json'), 'utf8')) as WordsManifest
  } catch {
    return fail(WORD_APPEND_WRITE)
  }
  const located = prepareWordAppend(manifest, null, raw)
  if (!located.ok) return fail(located.message)
  const dataRoot = resolve(root, 'public', 'data', 'words')
  const filePath = resolve(dataRoot, located.value.relativeFile)
  const fromRoot = relative(dataRoot, filePath)
  if (!isInsideWordsRoot(fromRoot)) return fail(WORD_APPEND_PATH)

  let existingText: string | null
  try {
    existingText = await readFile(filePath, 'utf8')
  } catch (error) {
    const code = error && typeof error === 'object' && 'code' in error ? error.code : ''
    if (code !== 'ENOENT') return fail(WORD_APPEND_WRITE)
    existingText = null
  }
  const prepared = prepareWordAppend(manifest, existingText, raw)
  if (!prepared.ok) return fail(prepared.message)
  await mkdir(dirname(filePath), { recursive: true })
  await writeFile(filePath, prepared.value.contents, 'utf8')
  return { status: 200, body: '{"ok":true}' }
}

function fail(message: string): { status: number; body: string } {
  return { status: 400, body: JSON.stringify({ message }) }
}

function readBody(req: IncomingMessage): Promise<string> {
  return new Promise((resolveBody, reject) => {
    const chunks: Buffer[] = []
    let size = 0
    let failed = false
    req.on('data', (chunk: Buffer) => {
      if (failed) return
      size += chunk.length
      if (size > MAX_BYTES) {
        failed = true
        reject(new Error('too large'))
        req.destroy()
        return
      }
      chunks.push(chunk)
    })
    req.on('end', () => {
      if (!failed) resolveBody(Buffer.concat(chunks).toString('utf8'))
    })
    req.on('error', (error: Error) => {
      if (!failed) reject(error)
    })
  })
}

function reply(res: ServerResponse, status: number, body: string): void {
  if (res.writableEnded) return
  res.statusCode = status
  res.setHeader('Content-Type', 'application/json; charset=utf-8')
  res.end(body)
}
```

```ts
import {
  WORD_APPEND_FAILED,
  WORD_SAVE_PATH,
  parseWordAppendBody,
  type WordAppendBody,
} from '@/utils/wordAppend'

/**
 * 把一条新词交给开发服务器，追加到对应课次 JSON。
 * 字段不合法时不发请求。接口失败时抛出服务器返回的原因。
 *
 * @param body 册、单元、课和五个词字段
 */
export async function saveNewWord(body: WordAppendBody): Promise<void> {
  const parsed = parseWordAppendBody(body)
  if (!parsed.ok) throw new Error(parsed.message)
  const res = await fetch(WORD_SAVE_PATH, {
    method: 'PUT',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(parsed.value),
  })
  if (res.ok) return
  let message = WORD_APPEND_FAILED
  try {
    const payload = (await res.json()) as { message?: unknown }
    if (typeof payload.message === 'string' && payload.message) message = payload.message
  } catch {
    message = WORD_APPEND_FAILED
  }
  throw new Error(message)
}
```
